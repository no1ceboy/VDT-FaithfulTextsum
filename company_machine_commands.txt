# VDT-FaithfulTextsum: canonical company-machine commands

# CASE 01 - CLEAN THE CANONICAL id/text/summary DATASET (original unchanged)
python3 -m src.data.clean_dataset \
  --input data/training.jsonl \
  --output data/training_cleaned.jsonl \
  --source_col text \
  --reference_col summary \
  --replacement_policy drop \
  --drop_stale_token_lengths \
  --report results/training_cleaning.json \
  --audit_log results/training_cleaning.audit.jsonl

# CASE 02 - DIRECT ATOMIC CLAIM EXTRACTION (local GPU smoke test)
# Remove --limit 1 for the full dataset. For id/text/summary data, use
# --source_col text --summary_col summary and change only the input path.
python3 -m src.evaluate.atomic_claims \
  --backend local \
  --model models/Llama-3.2-3B-Instruct \
  --input data/training_cleaned.jsonl \
  --source_col text \
  --summary_col summary \
  --output results/atomic_claims.local.jsonl \
  --coverage_output results/atomic_coverage.local.jsonl \
  --error_output results/atomic_errors.local.jsonl \
  --device cuda \
  --dtype bfloat16 \
  --model_batch_size 8 \
  --top_k 3 \
  --limit 1

# CASE 03 - JUDGE THE ATOMIC CLAIMS (local GPU)
python3 -m src.evaluate.llm_claim_judge \
  --backend local \
  --model models/Llama-3.2-3B-Instruct \
  --input results/atomic_claims.local.jsonl \
  --output results/atomic_verdicts.local.jsonl \
  --device cuda \
  --dtype bfloat16 \
  --model_batch_size 8 \
  --overwrite

# CASE 04 - AGGREGATE ATOMIC JUDGMENTS TO PARENT SUMMARIES
python3 -m src.evaluate.atomic_aggregate \
  --input results/atomic_verdicts.local.jsonl \
  --output results/parent_verdicts.local.jsonl

# CASE 05 - PREPARE TRAINING SPLITS (canonical id/text/summary format)
python3 scripts/prepare_grpo_data.py \
  --input data/training_cleaned.jsonl \
  --source_col text \
  --reference_col summary \
  --output_dir results/training_splits \
  --validation_fraction 0.1 \
  --test_fraction 0.1 \
  --seed 42

# CASE 06 - EXPERIMENT A: LORA + FACTCC + REFERENCE REWARD
python3 -m src.training.train_grpo \
  --model models/Llama-3.2-3B-Instruct \
  --train_jsonl results/training_splits/train.jsonl \
  --eval_jsonl results/training_splits/validation.jsonl \
  --output_dir outputs/grpo_lora_factcc_reference_001 \
  --faithfulness_metrics factcc \
  --factcc_model_path models/factcc \
  --reference_weight 0.25 \
  --num_generations 4 \
  --num_generations_eval 1 \
  --per_device_train_batch_size 1 \
  --per_device_eval_batch_size 1 \
  --gradient_accumulation_steps 4 \
  --max_steps 5000 \
  --max_completion_length 256 \
  --eval_strategy steps \
  --eval_steps 50 \
  --save_steps 50 \
  --logging_steps 10 \
  --report_to tensorboard \
  --run_name grpo-lora-factcc-reference-001 \
  --experiment_name grpo-faithfulness-ablation \
  --ablation lora-factcc-reference \
  --finetuning_method lora \
  --lora_r 16 \
  --lora_alpha 32 \
  --use_vllm \
  --vllm_gpu_memory_utilization 0.25 \
  --precision bf16 \
  --seed 42
